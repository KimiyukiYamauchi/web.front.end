## 1. `js/index.js` の実装イメージ

配列＋DOM操作の組み合わせです。第6回 4.2 の `index.html` と `css/styles.css` はそのまま使います（HTMLは変更しません）。

```js
let memoList = ["本を読む"]; // 初期データ

const addButton = document.getElementById("add-button");
const addText = document.getElementById("add-text");
const memoListEl = document.getElementById("memo-list");

addButton.addEventListener("click", () => {
  const text = addText.value.trim();
  if (text === "") return;

  memoList.push(text);
  addText.value = "";
  renderMemoList();
});

function renderMemoList() {
  memoListEl.innerHTML = "";

  memoList.forEach((memo, index) => {
    // ひな形と同じ <li><div><p>…</p><button>削除</button></div></li> の構造を作る
    const li = document.createElement("li");
    const div = document.createElement("div");
    const p = document.createElement("p");
    p.textContent = memo;

    const deleteButton = document.createElement("button");
    deleteButton.textContent = "削除";
    deleteButton.addEventListener("click", () => {
      memoList = memoList.filter((_, i) => i !== index);
      renderMemoList();
    });

    div.appendChild(p);
    div.appendChild(deleteButton);
    li.appendChild(div);
    memoListEl.appendChild(li);
  });
}

// 初期表示（ひな形に書かれていた「本を読む」を、配列の内容で描画し直す）
renderMemoList();
```
